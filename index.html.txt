/**
 * 純 Web Audio API 音效合成引擎 (0 Byte 外部依賴)
 */
class SoundFx {
  constructor() {
    this.ctx = null;
    this.enabled = true;
  }

  init() {
    if (!this.ctx) {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (AudioCtx) {
        this.ctx = new AudioCtx();
      }
    }
    if (this.ctx && this.ctx.state === 'suspended') {
      this.ctx.resume();
    }
  }

  // 答對的清脆升調雙音 (Ding-Dong / Sparkle)
  playCorrect() {
    if (!this.enabled || !this.ctx) return;
    const now = this.ctx.currentTime;
    const osc1 = this.ctx.createOscillator();
    const osc2 = this.ctx.createOscillator();
    const gain = this.ctx.createGain();

    osc1.type = 'triangle';
    osc2.type = 'sine';

    osc1.frequency.setValueAtTime(587.33, now); // D5
    osc1.frequency.exponentialRampToValueAtTime(880, now + 0.12); // A5

    osc2.frequency.setValueAtTime(880, now + 0.12);
    osc2.frequency.exponentialRampToValueAtTime(1174.66, now + 0.28); // D6

    gain.gain.setValueAtTime(0.18, now);
    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.4);

    osc1.connect(gain);
    osc2.connect(gain);
    gain.connect(this.ctx.destination);

    osc1.start(now);
    osc2.start(now + 0.1);
    osc1.stop(now + 0.35);
    osc2.stop(now + 0.45);
  }

  // 答錯時的溫和警示音
  playWrong() {
    if (!this.enabled || !this.ctx) return;
    const now = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();

    osc.type = 'sawtooth';
    osc.frequency.setValueAtTime(260, now);
    osc.frequency.linearRampToValueAtTime(190, now + 0.22);

    gain.gain.setValueAtTime(0.12, now);
    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.25);

    osc.connect(gain);
    gain.connect(this.ctx.destination);

    osc.start(now);
    osc.stop(now + 0.25);
  }

  // 點擊與轉場按鈕音
  playClick() {
    if (!this.enabled || !this.ctx) return;
    const now = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();

    osc.type = 'sine';
    osc.frequency.setValueAtTime(440, now);
    osc.frequency.exponentialRampToValueAtTime(660, now + 0.08);

    gain.gain.setValueAtTime(0.12, now);
    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.09);

    osc.connect(gain);
    gain.connect(this.ctx.destination);

    osc.start(now);
    osc.stop(now + 0.09);
  }

  // 最終通關勝利號角
  playVictory() {
    if (!this.enabled || !this.ctx) return;
    const notes = [523.25, 659.25, 783.99, 1046.50]; // C, E, G, High C
    notes.forEach((freq, i) => {
      const now = this.ctx.currentTime + (i * 0.11);
      const osc = this.ctx.createOscillator();
      const gain = this.ctx.createGain();

      osc.type = 'triangle';
      osc.frequency.setValueAtTime(freq, now);

      gain.gain.setValueAtTime(0.18, now);
      gain.gain.exponentialRampToValueAtTime(0.001, now + 0.35);

      osc.connect(gain);
      gain.connect(this.ctx.destination);

      osc.start(now);
      osc.stop(now + 0.38);
    });
  }
}

/**
 * 遊戲狀態管理（純記憶體模式）

let currentQuestionIndex = 0;
let score = 0;
let correctCount = 0;
let answered = false;

// DOM 元素快取
const startScreen = document.getElementById('startScreen');
const quizScreen = document.getElementById('quizScreen');
const endScreen = document.getElementById('endScreen');

const startBtn = document.getElementById('startBtn');

const soundToggle = document.getElementById('soundToggle');

const qNumberText = document.getElementById('qNumberText');
const scoreText = document.getElementById('scoreText');
const progressTrack = document.getElementById('progressTrack');
const questionTitle = document.getElementById('questionTitle');

// 初始化進度條 step
function buildProgressTrack() {
  progressTrack.innerHTML = '';
  for (let i = 0; i < QUIZ_DATA.length; i++) {
    const step = document.createElement('div');
    step.className = 'progress-step';
    step.id = `step-${i}`;
    step.innerText = (i + 1);
    progressTrack.appendChild(step);
  }
}

function updateProgressTrack() {
  for (let i = 0; i < QUIZ_DATA.length; i++) {
    const step = document.getElementById(`step-${i}`);
    if (!step) continue;
    step.classList.remove('active', 'done');
    if (i < currentQuestionIndex) {
      step.classList.add('done');
    } else if (i === currentQuestionIndex) {
      step.classList.add('active');
    }
  }
}

// 畫面切換
function showScreen(screen) {
  startScreen.classList.remove('active');
  quizScreen.classList.remove('active');
  endScreen.classList.remove('active');
  screen.classList.add('active');
}

// 載入題目
function loadQuestion(index) {
  answered = false;
  const data = QUIZ_DATA[index];
  
  qNumberText.innerText = `第 ${index + 1} / ${QUIZ_DATA.length} 關`;
  questionTitle.innerText = `Q${data.id}. ${data.question}`;
  
  // 隱藏反饋與下一題按鈕
  feedbackBubble.style.display = 'none';
  nextBtn.style.display = 'none';
  
  updateProgressTrack();

  // 渲染選項
  optionsGrid.innerHTML = '';
  const letters = ['A', 'B', 'C', 'D'];
  data.options.forEach((optText, optIndex) => {
    const btn = document.createElement('button');
    btn.className = 'option-btn';
    btn.innerHTML = `
      <span class="opt-tag">${letters[optIndex]}</span>
      <span style="flex:1;">${optText}</span>
    `;
    btn.addEventListener('click', () => handleOptionClick(optIndex));
    optionsGrid.appendChild(btn);
  });
}

// 選項點擊判定
function handleOptionClick(selectedIndex) {
  if (answered) return;
  answered = true;

  const currentQ = QUIZ_DATA[currentQuestionIndex];
  const buttons = optionsGrid.querySelectorAll('.option-btn');
  const isCorrect = selectedIndex === currentQ.correctAnswer;

  // 鎖定所有按鈕並標示狀態
  buttons.forEach((btn, idx) => {
    btn.disabled = true;
    if (idx === currentQ.correctAnswer) {
      btn.classList.add('correct');
    } else if (idx === selectedIndex) {
      btn.classList.add('wrong');
    }
  });

  // 計分與音效
  if (isCorrect) {
    sfx.playCorrect();
    score += 100;
    correctCount++;
    scoreText.innerText = score;
    bubbleHeader.innerHTML = '🦉 <span style="color: #2D6A4F;">貓頭鷹部長點頭讚許：答對了！金幣 +100 🌻</span>';
  } else {
    sfx.playWrong();
    bubbleHeader.innerHTML = '🦉 <span style="color: #BA3C2A;">貓頭鷹部長遞上筆記：別灰心，記下重點！</span>';
  }

  // 顯示溫暖解析
  bubbleText.innerText = currentQ.explanation;
  feedbackBubble.style.display = 'block';

  // 調整下一題按鈕文字
  if (currentQuestionIndex === QUIZ_DATA.length - 1) {
    nextBtn.innerHTML = '<span>結算考核成績</span> 🏆';
  } else {
    nextBtn.innerHTML = '<span>前進下一關</span> ➔';
  }
  nextBtn.style.display = 'inline-flex';
}

// 結算畫面
function showSummary() {
  sfx.playVictory();
  showScreen(endScreen);

  finalCorrectCount.innerText = correctCount;
  finalScore.innerText = score;

  if (correctCount === 8) {
    badgeIcon.innerText = '🌟';
    rankTitle.innerText = '金牌傳奇實習生';
    rankFeedback.innerText = '完全掌握了電通年末考核的核心概念！主管貓頭鷹部長為你起立鼓掌！';
  } else if (correctCount >= 6) {
    badgeIcon.innerText = '🏅';
    rankTitle.innerText = '傑出潛力之星';
    rankFeedback.innerText = '表現非常亮眼！對考核重點與自我成長有很高的敏銳度！';
  } else if (correctCount >= 4) {
    badgeIcon.innerText = '🌱';
    rankTitle.innerText = '合格成長夥伴';
    rankFeedback.innerText = '基本概念都掌握了！記得自評送出前再回顧一次要點喔！';
  } else {
    badgeIcon.innerText = '🐾';
    rankTitle.innerText = '森林實習再接再厲';
    rankFeedback.innerText = '別氣餒，複習一下貓頭鷹部長的小筆記，再來挑戰一次吧！';
  }
}

// 重新開始遊戲（狀態完全重置）
function restartGame() {
  currentQuestionIndex = 0;
  score = 0;
  correctCount = 0;
  answered = false;
  scoreText.innerText = '0';
  loadQuestion(0);
  showScreen(quizScreen);
}
